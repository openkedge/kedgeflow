from pathlib import Path
import datetime
import gzip
import hashlib
import json
import re
import runpy
import shutil
import zipfile

root = Path.cwd()
base = root / '.cache/physical-operating-freeze-20261023'
out = root / 'results/verification/rfc-m8-physical-operating-records-2026-10-23'
assert not (out / 'final-source.json').exists(), 'finalize exactly once'
h = json.loads((base / 'source-sha256.json').read_text())
source_hashes = runpy.run_path('scripts/conformance.py')['source_hashes']
assert source_hashes() == h
s = json.loads((out / 'summary.json').read_text())
assert s['status'] == 'passed_software_checks' and s['source_sha256'] == h
assert all(c['exit_code'] == 0 for c in s['commands'])
r = s['rust_tests']
assert r['failed'] == r['ignored'] == 0 and r['passed'] == r['manifest_cases']
assert len([c for c in s['commands'] if c['name'].startswith('artifact-verify-')]) == 14
assert len([c for c in s['commands'] if c['name'].startswith('operating-verify-')]) == 4
with zipfile.ZipFile(base / 'source.zip') as z:
    assert {n:hashlib.sha256(z.read(n)).hexdigest() for n in z.namelist()} == h
shutil.copyfile(base / 'source.zip', out / 'source.zip')
inputs = json.loads((base / 'historical-inputs.json').read_text())
assert len(inputs['sha256']) == 8
for name, pin in inputs['sha256'].items():
    assert hashlib.sha256((root / name).read_bytes()).hexdigest() == pin
shutil.copyfile(base / 'historical-inputs.json', out / 'historical-inputs.json')
artifacts = out / 'native-artifacts'
index = json.loads((artifacts / 'kube-drain-fixtures/index.json').read_text())
assert index['status'] == 'passed' and len(index['executed']) == 11
assert {e['case'] for e in index['executed']} == set(index['required_cases'])
for e in index['executed']:
    assert e['exit_code'] == 0
    for name, pin in e['public_sha256'].items():
        assert hashlib.sha256((artifacts / 'kube-drain-fixtures' / e['public_directory'] / name).read_bytes()).hexdigest() == pin
operating_summary = {}
for name, expected in [('unknown','unresolved'),('completed','observed_completed'),('no-effect','no_physical_effect'),('no-dispatch','no_physical_dispatch')]:
    path = artifacts / 'physical-operating' / name
    bundle = json.loads((path / 'bundle.json').read_text())
    trust = json.loads((path / 'trust.json').read_text())
    verification = json.loads((path / 'verification.json').read_text())
    # Cargo's separate verifier output follows compiler diagnostics; compare its
    # JSON tail to the in-test shipping process result.
    text = (out / f'operating-verify-{name}.log').read_text()
    cli = json.loads(text[text.find('{'):])
    assert cli == verification
    assert verification['observation']['resolution'] == expected and verification['execution_authority'] is False
    assert verification['stop_required'] is True
    assert verification['deadline'] == bundle['history']['original']['dispatched_at'] + bundle['history']['policy']['reconcile_for_ms']
    assert bundle['checkpoint']['event_count'] == len(bundle['history']['events']) == verification['event_count']
    assert bundle['checkpoint']['enrollment_digest'] == trust['enrollment_digest']
    if name == 'unknown':
        assert verification['observation']['native_status'] == 'committed'
        assert verification['after_queries'] == 1 and verification['before_queries'] == 0
        assert verification['pending'] is None and verification['incident'] is not None
        assert verification['incident']['resource_owner_id'] == 'team:platform-owner'
        assert verification['incident']['on_call_team_id'] == 'team:platform-oncall'
    else:
        assert verification['before_queries'] == verification['after_queries'] == 0
    operating_summary[name] = {'resolution':expected,'events':verification['event_count'],'stop_required':True,'execution_authority':False}
(out / 'operating-classifications.json').write_text(json.dumps(operating_summary, indent=2) + '\n')
focused = []
for p in sorted(root.glob('.cache/physical-operating-*.log')):
    if p.name.startswith('physical-operating-conformance-'):
        continue
    raw = p.read_bytes()
    name = 'diagnostic-' + p.name + '.gz'
    (out / name).write_bytes(gzip.compress(raw, mtime=0))
    focused.append({'file':name,'uncompressed_sha256':hashlib.sha256(raw).hexdigest(),
                    'scope':'development diagnostic; may precede repairs; not final-source conformance'})
(out / 'focused-attempts.json').write_text(json.dumps(focused, indent=2) + '\n')
observations = json.loads((base / 'observations.json').read_text())
observations.append({'at':datetime.datetime.now(datetime.timezone.utc).isoformat(),'stage':'after_required_conformance','unchanged':True,'source_files':len(h)})
(out / 'source-observations.json').write_text(json.dumps(observations, indent=2) + '\n')
(out / 'requirements-audit.json').write_text(json.dumps({
    'scope':'RFC0056 O1 closed physical operating record/replay prerequisite; no managed action enablement, live CAS, polling or incident delivery',
    'qualified':[
        'Explicit closed versioned safe-integer policy with pinned owner/on-call/incident sink, query/backoff/delivery limits and finite post-deadline observation',
        'Original native dispatch/grant/attempt/root/family/contract/resource/realm/node correlation; deadline fixed to retained original dispatch even if acceptance is ambiguous',
        'Complete signed original physical aggregate verification, separately pinned operating checkpoint role and explicit native-committed versus physical-unresolved classification',
        'Hash-linked numbered query plans; pending survives serialized restart and interrupted/malformed/transport results keep the slot consumed',
        'Full 64 pre-deadline plus 64 post-deadline query budget replay across 258 events, bounded capped backoff, safe timestamp overflow refusal and no budget reset',
        'Owned stable escalation/outbox intent precedes post-deadline observation; a late checkpoint cannot hide an overdue unresolved event prefix',
        'Historical completion, positive no-dispatch and accepted no-effect classifications from actual immutable prior signed physical proof inputs',
        'Historical-proof cache includes complete proof/trust/policy bytes, has fixed entry/byte limits, and never caches deadlines, query counters or live decisions',
        'Shipping bounded-input offline verifier, missing/null/duplicate/unknown/numeric/role/signature/correlation/time/budget/chain/evidence substitution negatives',
        'Four new separate operating offline replays plus fourteen prior family/physical replays; complete native/Python/Node/SQL/real-cluster regression corpus',
    ],
    'not_qualified_or_incomplete':[
        'O2 protected managed descriptor/native/target/aggregate enrollment and durable expected-head CAS before query I/O',
        'O2 actual deadline/stop enforcement, concurrent-worker conservation, owned incident sink delivery/deduplication and live shipping crash/restart',
        'O3 current SQL reservation/fence evidence under declared storage generation and covered/external writer boundary; historical native commit alone is insufficient',
        'O4 combined managed controller/supervisor/recovery/official SDK software qualification; O1 reuses historical physical aggregates, not new managed physical effects',
        'Independent custody/writers, production maintenance, physical node failure, workload SLO, whole-saga/authority-peer and release acceptance',
    ],
    'execution_authority_from_record':False,
    'broader_goal_complete':False,
}, indent=2) + '\n')
(out / 'qualification-build.json').write_text(json.dumps({
    'CARGO_PROFILE_DEV_DEBUG':'0','CARGO_PROFILE_TEST_DEBUG':'0','CARGO_INCREMENTAL':'0','CARGO_BUILD_JOBS':'2',
    'topology':'Darwin ARM64; actual peer PostgreSQL 18.3, three etcd 3.6.5 members, three Kubernetes v1.37.0 container nodes sharing one isolated task-owned VZ VM',
    'operating_record_tests':'Pure deterministic historical record/replay and actual shipping offline command; no managed controller, live CAS, incident network or new physical SLO claim',
    'historical_inputs':'Eight immutable aggregate/trust files from the d80296e qualification record; exact hashes in tests and historical-inputs.json',
    'clocks':'Historical source timestamps for O1 offline records; existing real-clock live physical regressions retain original authority and transport windows',
    'scope':'Local software regression and record prerequisite only; all managed operating/deployment acceptance remains open',
}, indent=2) + '\n')
python_count = int(re.search(r'(\d+) passed', (out / 'python.log').read_text()).group(1))
(out / 'README.md').write_text(f'''# Physical operating record prerequisite — 23–24 October 2026

(out / 'final-source.json').write_text(json.dumps({'profile':'KF-FINAL-SOURCE-0.1-draft','status':'passed_final_source_unchanged','baseline_commit':s['baseline_commit'],'source_sha256':h,'historical_input_sha256':inputs['sha256'],'evidence_sha256':evidence,'observations':observations},indent=2) + '\n')
