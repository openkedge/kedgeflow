from pathlib import Path
import datetime
import gzip
import hashlib
import json
import re
import runpy

root = Path.cwd()
base = root / '.cache/kube-sdk-freeze-20261023'
out = root / 'results/verification/rfc-m8-kubernetes-sdk-2026-10-23'
assert not (out / 'final-source.json').exists(), 'finalize exactly once'
h = json.loads((base / 'source-sha256.json').read_text())
source_hashes = runpy.run_path('scripts/conformance.py')['source_hashes']
assert source_hashes() == h
s = json.loads((out / 'summary.json').read_text())
assert s['status'] == 'passed_software_checks' and s['source_sha256'] == h
assert all(c['exit_code'] == 0 for c in s['commands'])
r = s['rust_tests']
assert r['failed'] == r['ignored'] == 0 and r['passed'] == r['manifest_cases']
assert len([c for c in s['commands'] if c['name'].startswith('artifact-verify-')]) == 14
with zipfile.ZipFile(base / 'source.zip') as z:
    assert {n: hashlib.sha256(z.read(n)).hexdigest() for n in z.namelist()} == h
shutil.copyfile(base / 'source.zip', out / 'source.zip')
artifacts = out / 'native-artifacts'
index = json.loads((artifacts / 'kube-drain-fixtures/index.json').read_text())
assert index['status'] == 'passed' and len(index['executed']) == 11
assert {e['case'] for e in index['executed']} == set(index['required_cases'])
for e in index['executed']:
    assert e['exit_code'] == 0
    for name, pin in e['public_sha256'].items():
        assert hashlib.sha256((artifacts / 'kube-drain-fixtures' / e['public_directory'] / name).read_bytes()).hexdigest() == pin
for name, expected in [('completed', 'forward_completed'), ('unknown', 'unresolved')]:
    path = artifacts / ('kube-drain-sdk-' + name)
    wire = json.loads((path / 'wire.json').read_text())
    assert wire['native_adapter_dispatches'] == wire['native_broker_mints'] == 1
    assert wire['provider_baselines'] == wire['provider_collections'] == 1
    assert wire['shipping_recovery'] == {'deferred':0,'examined':1,'next_cursor':None,'reported':1,'unavailable':0}
    assert wire['traces'][0]['physical_status'] == 'unresolved'
    assert wire['traces'][0]['state'] == ('accepted' if name == 'completed' else 'unknown')
    assert wire['traces'][0]['native_state'] == ('committed' if name == 'completed' else 'unknown')
    methods = [w['method'] for w in wire['traces'][0]['wire']]
    assert methods.count('admit') == methods.count('issue') == 1
    assert sum(w['grantPresent'] for w in wire['traces'][0]['wire']) == 1
    if name == 'unknown':
        original, resume = wire['traces']
        assert resume['attempt_id'] == original['attempt_id'] and resume['grant_digest'] == original['grant_digest']
        assert resume['state'] == 'accepted' and resume['native_state'] == 'committed' and resume['physical_status'] == 'unresolved'
        assert [w['method'] for w in resume['wire']] == ['server/discover','kedgeflow/status','kedgeflow/status']
        assert not any(w['grantPresent'] for w in resume['wire'])
    verification = json.loads((path / 'verification.json').read_text())
    assert verification['status'] == expected and not verification['execution_authority']
    assert (verification['held_forward'], verification['held_compensation']) == (2, 1)
